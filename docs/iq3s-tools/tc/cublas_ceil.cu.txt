// cuBLAS ceiling for FNFL's prompt products: C[M][N] = A[M][K] . W[N][K]^T, FP16 in, FP32 accumulate, best of the
// tensor-op algorithms; also batched for the experts (uniform rows an expert) and HC down (4 streams)
#include <cstdio>
#include <vector>
#include <cublas_v2.h>
#include <cuda_fp16.h>
#define CK(x) do{auto e=(x); if(e){printf("%d: err %d\n",__LINE__,(int)e); return 1;}}while(0)
cublasHandle_t h;
float timeit(int M,int N,int K,int batch,cublasGemmAlgo_t algo,bool c32, void*A, void*W, void*C, long sA, long sW, long sC){
  float alpha=1,beta=0; cudaEvent_t e0,e1; cudaEventCreate(&e0); cudaEventCreate(&e1);
  auto run=[&](){ return batch==1 ? cublasGemmEx(h, CUBLAS_OP_T, CUBLAS_OP_N, N, M, K, &alpha, W, CUDA_R_16F, K, A, CUDA_R_16F, K, &beta, C, c32?CUDA_R_32F:CUDA_R_16F, N, CUBLAS_COMPUTE_32F, algo)
    : cublasGemmStridedBatchedEx(h, CUBLAS_OP_T, CUBLAS_OP_N, N, M, K, &alpha, W, CUDA_R_16F, K, sW, A, CUDA_R_16F, K, sA, &beta, C, c32?CUDA_R_32F:CUDA_R_16F, N, sC, batch, CUBLAS_COMPUTE_32F, algo); };
  if (run()) return -1; for(int i=0;i<3;++i) run();
  cudaEventRecord(e0); for(int i=0;i<20;++i) run(); cudaEventRecord(e1); cudaEventSynchronize(e1);
  float ms; cudaEventElapsedTime(&ms,e0,e1); return ms/20; }
int main(){
  CK(cublasCreate(&h)); CK(cublasSetMathMode(h, CUBLAS_TENSOR_OP_MATH));
  struct S{const char*n; int M,N,K,batch; float fnfl_ms;} sh[]={
    {"GDN qkv+gate",4096,8192,2560,1,3.291},{"QSA q+k+v",4096,6656,2560,1,2.834},{"out 2560x3072",4096,2560,3072,1,0},
    {"shexp down 2560x320",4096,2560,320,1,0},{"shexp gate+up",4096,640,2560,1,0.413},{"HC up",4096,10240,320,1,1.429},
    {"HC down (4 streams)",4096,320,2560,4,0.602},{"experts gate+up (80 rows x512)",80,640,2560,512,3.161},{"experts down (80 rows x512)",80,2560,320,512,1.871}};
  size_t big=size_t(1)<<29, wbig=size_t(1)<<31; void *A,*W,*C; CK(cudaMalloc(&A,big)); CK(cudaMalloc(&W,wbig)); CK(cudaMalloc(&C,big*2));
  { std::vector<__half> hv(big/2); for(size_t i=0;i<hv.size();++i) hv[i]=__float2half(((i*2654435761u)%2001)/2000.0f-0.5f);
    cudaMemcpy(A,hv.data(),big,cudaMemcpyHostToDevice); for(int q=0;q<4;++q) cudaMemcpy((char*)W+q*big,hv.data(),big,cudaMemcpyHostToDevice); }
  printf("%-34s %10s %10s %9s %9s %8s\n","product","best ms","algo","TFLOP/s","FNFL ms","FNFL/cuBLAS");
  for(auto&s:sh){ float best=1e9; int ba=-2; for(int a=-1;a<=15;++a){ cublasGemmAlgo_t al = a<0?CUBLAS_GEMM_DEFAULT_TENSOR_OP:(cublasGemmAlgo_t)(CUBLAS_GEMM_ALGO0_TENSOR_OP+a);
      float t=timeit(s.M,s.N,s.K,s.batch,al,true,A,W,C,long(s.M)*s.K,long(s.N)*s.K,long(s.M)*s.N); if(t>0&&t<best){best=t;ba=a;} }
    double fl=2.0*s.M*s.N*s.K*s.batch; printf("%-34s %10.3f %10d %9.1f %9.3f %8.2f\n",s.n,best,ba,fl/best/1e9,s.fnfl_ms,s.fnfl_ms>0?s.fnfl_ms/best:0); }
}
