// the RidgeFill map's cost per card: [rows x 10240] x [10240 x 1280] FP16 -> F32-accumulate, cuBLAS (an upper bound
// on how fast an engine kernel could do it), rows 4096 (a chunk half)
#include <cstdio>
#include <cublas_v2.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>
int main() {
  cudaSetDevice(0); cublasHandle_t h; cublasCreate(&h);
  const int M = 4096, K = 10240, N = 1280;
  half *a, *b, *c; cudaMalloc(&a, size_t(M) * K * 2); cudaMalloc(&b, size_t(K) * N * 2 * 8); cudaMalloc(&c, size_t(M) * N * 2);
  cudaMemset(a, 0, size_t(M) * K * 2); cudaMemset(b, 0, size_t(K) * N * 16);
  const float one = 1.f, zero = 0.f; cudaEvent_t e0, e1; cudaEventCreate(&e0); cudaEventCreate(&e1);
  for (int w = 0; w < 2; ++w) {
    cudaEventRecord(e0);
    for (int i = 0; i < 48; ++i)
      cublasGemmEx(h, CUBLAS_OP_N, CUBLAS_OP_N, N, M, K, &one, b + size_t(i % 8) * K * N, CUDA_R_16F, N, a, CUDA_R_16F, K, &zero, c, CUDA_R_16F, N, CUBLAS_COMPUTE_32F, CUBLAS_GEMM_DEFAULT_TENSOR_OP);
    cudaEventRecord(e1); cudaEventSynchronize(e1);
    float ms; cudaEventElapsedTime(&ms, e0, e1);
    if (w) printf("map GEMM %dx%dx%d: %.3f ms each, %.1f TFLOP/s\n", M, K, N, ms / 48, 2.0 * M * K * N / (ms / 48 * 1e-3) / 1e12);
  }
}
